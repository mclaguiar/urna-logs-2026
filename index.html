turnos = FileAttachment("data/migracao/turnos.csv").csv({typed: true})
painel = FileAttachment("data/painel_uf.csv").csv({typed: true})
achadosAb = FileAttachment("data/achados_ab.csv").csv({typed: true})
votos = (turno, grupo) => turnos
  .filter(d => +d.turno === turno && String(d.grupo) === String(grupo))
  .reduce((s, d) => s + (+d.votos || 0), 0)
validos = turno => turno === 1
  ? votos(1, "13") + votos(1, "22") + votos(1, "eliminados")
  : votos(2, "13") + votos(2, "22")
fmt = n => (n == null || Number.isNaN(+n)) ? "—" : Math.round(+n).toLocaleString("pt-BR")
pct = x => (x == null || !isFinite(+x)) ? "—" : (100 * +x).toLocaleString("pt-BR", {minimumFractionDigits: 1, maximumFractionDigits: 1}) + "%"
deltaTxt = (depois, antes) => {
  const d = depois - antes
  return (d > 0 ? "+" : "") + fmt(d)
}
rows = painel.map(d => ({
  secoes: d.secoes_1t == null || d.secoes_1t === "" ? 0 : +d.secoes_1t,
  div1: d.secoes_divergentes_1t == null || d.secoes_divergentes_1t === "" ? 0 : +d.secoes_divergentes_1t,
  div2: d.secoes_divergentes_2t == null || d.secoes_divergentes_2t === "" ? 0 : +d.secoes_divergentes_2t,
  a: d.achados_A == null || d.achados_A === "" ? 0 : +d.achados_A,
  b: d.achados_B == null || d.achados_B === "" ? 0 : +d.achados_B
}))
mSec = rows.reduce((s, d) => s + d.secoes, 0)
sumA = rows.reduce((s, d) => s + d.a, 0)
sumB = rows.reduce((s, d) => s + d.b, 0)
divOficial = rows.reduce((s, d) => s + d.div1 + d.div2, 0)
fecha = mSec - sumA
fichaA = achadosAb.filter(d => String(d.nivel).toUpperCase() === "A")
linkSecao = jk => {
  const p = String(jk).split(":")
  return `secao.html?uf=${p[0]}&mun=${+p[1]}&zona=${+p[2]}&secao=${+p[3]}`
}
cartao = (classe, rotulo, numero, sub) => html`<div class="kpi ${classe}"><div class="rot">${rotulo}</div><div class="num">${numero}</div><div class="sub">${sub}</div></div>`
html`<div class="kpi-grid">
  ${cartao("lula", "Lula (13)", `${fmt(votos(1, "13"))} → ${fmt(votos(2, "13"))}`, `${deltaTxt(votos(2, "13"), votos(1, "13"))} votos · ${pct(votos(1, "13") / validos(1))} dos válidos no 1º · ${pct(votos(2, "13") / validos(2))} no 2º`)}
  ${cartao("flavio", "Flávio (22)", `${fmt(votos(1, "22"))} → ${fmt(votos(2, "22"))}`, `${deltaTxt(votos(2, "22"), votos(1, "22"))} votos · ${pct(votos(1, "22") / validos(1))} dos válidos no 1º · ${pct(votos(2, "22") / validos(2))} no 2º`)}
</div>`
html`<p><span class="selo">RDV ≡ BU ≡ CSV oficial em ${fmt(fecha)} de ${fmt(mSec)} seções</span></p>
<p>BU ≡ CSV oficial em todas as seções conferidas nos dois turnos (${fmt(divOficial)} divergências). ${sumA
  ? `A cadeia quebra em ${fmt(sumA)} seção${sumA === 1 ? "" : "ões"}, em que RDV ≠ BU.`
  : "RDV, BU e CSV oficial fecham em todas as seções."} Nível B: <strong>${fmt(sumB)}</strong> seções com dois ou mais sinais no extremo da UF.</p>
${fichaA.length
  ? html`<p>${fichaA.map(d => html`<a href="${linkSecao(d.join_key)}">Achado A · ${d.join_key}</a>`)}</p>`
  : ""}`
